# qr headler

✅ OverlayEditor z checkboxem:

- Zmienna `showQR` kontroluje `&qr=true/false`
- Podgląd w `<iframe>` odświeża się automatycznie

✅ Overlay obsługuje teraz QR:

- Gdy `?qr=true` → pokazuje kod w prawym dolnym rogu
- Automatycznie generowany dla `creatorId`
- Styl dopasowany (złoty na turkusie, z cieniem)

// app/overlay/page.tsx
'use client';

import { useSearchParams } from 'next/navigation';

export default function OverlayPage() {
const searchParams = useSearchParams();
const creatorId = searchParams.get('creatorId') || 'me';
const showQR = searchParams.get('qr') === 'true';

return (
<div className="relative w-full h-screen bg-transparent">
{/* Przykładowy alert napiwku lub feed */}
<div className="absolute top-10 left-10 text-white text-2xl bg-black/50 px-4 py-2 rounded-lg">
Nowy tip od @fan123: 5 USDC 💸
</div>

```
  {showQR && (
    <img
      src={`/api/qr?handle=${creatorId}`}
      alt="QR Tip Link"
      className="absolute bottom-4 right-4 w-24 h-24 bg-white p-1 rounded shadow-xl"
    />
  )}
</div>

```

);
}
