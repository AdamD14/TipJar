};

// === Główny komponent aplikacji ===

export default function SocialPostGenerator() {

```
// Stan do zarządzania aktywną zakładką

const [activeTab, setActiveTab] = useState('post-generator');

// Stany dla wszystkich generatorów

const [topic, setTopic] = useState('');

const [platforms, setPlatforms] = useState({ Twitter: false, Instagram: false, Facebook: false });

const [results, setResults] = useState([]);

const [isLoading, setIsLoading] = useState(false);

const [error, setError] = useState('');

// Funkcja do obsługi zmiany zakładki

const handleTabChange = (tab) => {

    setActiveTab(tab);

    // Resetowanie stanu przy zmianie zakładki

    setTopic('');

    setResults([]);

    setError('');

    setPlatforms({ Twitter: false, Instagram: false, Facebook: false });

};

// Funkcja do obsługi zmiany checkboxów

const handlePlatformChange = (e) => {

    const { name, checked } = e.target;

    setPlatforms(prev => ({ ...prev, [name]: checked }));

};

// Główna funkcja do wywoływania API Gemini

const callGeminiAPI = async (prompt) => {

    // W Next.js najlepiej jest stworzyć dedykowany endpoint API,

    // aby ukryć klucz API i logikę po stronie serwera.

    // Poniżej uproszczone wywołanie po stronie klienta dla celów demonstracyjnych.

    const apiKey = ""; // Klucz nie jest wymagany dla modelu 'flash'

    const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${apiKey}`;

    const payload = { contents: [{ role: "user", parts: [{ text: prompt }] }] };

    const response = await fetch(apiUrl, {

        method: 'POST',

        headers: { 'Content-Type': 'application/json' },

        body: JSON.stringify(payload)

    });

    if (!response.ok) {

        const errorBody = await response.text();

        throw new Error(`Błąd API: ${response.status} - ${errorBody}`);

    }

    const result = await response.json();

    const text = result.candidates?.[0]?.content?.parts?.[0]?.text;

    if (text) {

        return text;

    } else {

        throw new Error('API nie zwróciło poprawnej odpowiedzi.');

    }

};

// Funkcja do obsługi generowania

const handleSubmit = async (e) => {

    e.preventDefault();

    if (!topic) {

        setError('Proszę wpisać temat.');

        return;

    }

    setIsLoading(true);

    setResults([]);

    setError('');

    try {

        if (activeTab === 'post-generator') {

            const selectedPlatforms = Object.keys(platforms).filter(p => platforms[p]);

            if (selectedPlatforms.length === 0) {

                setError('Proszę wybrać co najmniej jedną platformę.');

                setIsLoading(false);

                return;

            }

            const generatedPosts = await Promise.all(

                selectedPlatforms.map(async (platform) => {

                    const prompt = `Jesteś ekspertem od marketingu w mediach społecznościowych. Stwórz profesjonalny, gotowy do publikacji post na platformę ${platform} o temacie: "${topic}". Trzymaj się wytycznych dla tej platformy (limit znaków, hashtagi, ton). Zwróć tylko treść posta, bez dodatkowych komentarzy.`;

                    const content = await callGeminiAPI(prompt);

                    return { title: platform, content };

                })

            );

            setResults(generatedPosts);

        } else if (activeTab === 'hashtag-generator') {

            const prompt = `Jesteś ekspertem od SEO i mediów społecznościowych. Wygeneruj listę 15-20 trafnych hashtagów dla tematu: "${topic}". Podziel je na 3 kategorie: popularne, tematyczne i niszowe. Formatuj odpowiedź w przejrzysty sposób. Zwróć tylko listę.`;

            const content = await callGeminiAPI(prompt);

            setResults([{ title: `Hashtagi dla: ${topic}`, content }]);

        } else if (activeTab === 'image-ideas') {

            const prompt = `Jesteś kreatywnym dyrektorem artystycznym. Wymyśl 3 unikalne i angażujące pomysły na zdjęcie lub grafikę do posta o temacie: "${topic}". Opisz każdy pomysł szczegółowo (kompozycja, kolory, nastrój). Formatuj odpowiedź jako listę punktowaną. Zwróć tylko listę pomysłów.`;

            const content = await callGeminiAPI(prompt);

            setResults([{ title: `Pomysły na grafikę dla: ${topic}`, content }]);

        }

    } catch (err) {

        setError(err.message);

        console.error(err);

    } finally {

        setIsLoading(false);

    }

};

// Dane do renderowania zakładek i formularzy

const tabsData = {

    'post-generator': { title: 'Generator Postów', buttonText: 'Generuj Posty', buttonClass: 'bg-indigo-600 hover:bg-indigo-700 disabled:bg-indigo-400 focus:ring-indigo-500' },

    'hashtag-generator': { title: 'Generator Hashtagów ✨', buttonText: 'Znajdź Hashtagi ✨', buttonClass: 'bg-teal-600 hover:bg-teal-700 disabled:bg-teal-400 focus:ring-teal-500' },

    'image-ideas': { title: 'Pomysły na Grafikę ✨', buttonText: 'Generuj Pomysły ✨', buttonClass: 'bg-purple-600 hover:bg-purple-700 disabled:bg-purple-400 focus:ring-purple-500' }

};

return (

    <div className="w-full max-w-3xl mx-auto bg-white rounded-2xl shadow-lg font-sans">

        <div className="p-6 md:p-8">

            <div className="text-center mb-6">

                <h1 className="text-3xl md:text-4xl font-bold text-gray-800">Kreator Treści AI</h1>

                <p className="text-gray-500 mt-2">Twórz posty, znajduj hashtagi i generuj pomysły z Gemini.</p>

            </div>

            {/* Nawigacja z zakładkami */}

            <div className="border-b border-gray-200">

                <nav className="-mb-px flex space-x-6" aria-label="Tabs">

                    {Object.keys(tabsData).map(tabKey => (

                        <button

                            key={tabKey}

                            onClick={() => handleTabChange(tabKey)}

                            className={`border-transparent whitespace-nowrap py-4 px-1 border-b-2 font-medium text-sm transition-colors ${activeTab === tabKey ? 'border-indigo-500 text-indigo-600' : 'text-gray-500 hover:text-gray-700 hover:border-gray-300'}`}

                        >

                            {tabsData[tabKey].title}

                        </button>

                    ))}

                </nav>

            </div>

            {/* Formularz */}

            <form onSubmit={handleSubmit} className="mt-8">

                <div className="space-y-6">

                    <div>

                        <label htmlFor="topic-input" className="block text-sm font-medium text-gray-700 mb-1">

                            {activeTab === 'post-generator' ? 'Temat posta:' : 'Temat:'}

                        </label>

                        <input

                            type="text"

                            id="topic-input"

                            value={topic}

                            onChange={(e) => setTopic(e.target.value)}

                            className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500 transition"

                            placeholder="Wpisz temat..."

                        />

                    </div>

                    {activeTab === 'post-generator' && (

                        <div>

                            <span className="block text-sm font-medium text-gray-700 mb-2">Wybierz platformy:</span>

                            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">

                                {Object.keys(platforms).map(p => (

                                    <label key={p} className="flex items-center p-3 border border-gray-300 rounded-lg cursor-pointer hover:bg-gray-50 has-[:checked]:bg-indigo-50 has-[:checked]:border-indigo-500 transition">

                                        <input type="checkbox" name={p} checked={platforms[p]} onChange={handlePlatformChange} className="h-5 w-5 rounded border-gray-300 text-indigo-600 focus:ring-indigo-500" />

                                        <span className="ml-3 text-sm font-medium text-gray-700">{p}</span>

                                    </label>

                                ))}

                            </div>

                        </div>

                    )}

                    <button type="submit" disabled={isLoading} className={`w-full flex items-center justify-center text-white font-semibold py-3 px-6 rounded-lg focus:outline-none focus:ring-2 focus:ring-opacity-50 transition-all duration-300 ${tabsData[activeTab].buttonClass}`}>

                        {isLoading ? <LoadingSpinner /> : <span>{tabsData[activeTab].buttonText}</span>}

                    </button>

                </div>

            </form>

        </div>

        {/* Wyniki i Błędy */}

        <div className="px-6 md:px-8 pb-8">

             {error && <div className="mt-6 text-center text-red-600 bg-red-100 p-3 rounded-lg">{error}</div>}

            <div className="mt-6 space-y-6">

                {results.map((result, index) => (

                    <ResultCard key={index} title={result.title} content={result.content} isError={false} />

                ))}

            </div>

        </div>

    </div>

);

```
